cmake_minimum_required(VERSION 3.22)
project(hudiy_car_runtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Network Multimedia WebEngineWidgets Bluetooth)
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavcodec libavutil libswscale)
pkg_check_modules(USB REQUIRED IMPORTED_TARGET libusb-1.0)
# A separately licensed, pinned open-source protocol dependency. No Hudiy files.
include(FetchContent)
set(OAA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(oaa_source
    GIT_REPOSITORY https://github.com/mrmees/openauto-prodigy.git
    GIT_TAG 3afc5dd8bbacb5ce68e109189b18b14b251fabef
    GIT_SUBMODULES libs/prodigy-oaa-protocol/proto
    SOURCE_SUBDIR libs/prodigy-oaa-protocol)
FetchContent_MakeAvailable(oaa_source)
add_executable(hudiy-car-runtime main.cpp usb_transport.cpp usb_transport.hpp)
target_link_libraries(hudiy-car-runtime PRIVATE prodigy-oaa-protocol
    Qt6::Widgets Qt6::Multimedia Qt6::WebEngineWidgets Qt6::Bluetooth PkgConfig::AV PkgConfig::USB)
target_compile_options(hudiy-car-runtime PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS hudiy-car-runtime RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
    add_executable(hudiy-notice-overlay-test test_notice_overlay.cpp)
    target_link_libraries(hudiy-notice-overlay-test PRIVATE Qt6::Widgets)
    target_compile_options(hudiy-notice-overlay-test PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME notification-overlay COMMAND hudiy-notice-overlay-test)
    set_tests_properties(notification-overlay PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(hudiy-audio-dsp-test test_audio_dsp.cpp)
    target_compile_options(hudiy-audio-dsp-test PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME projection-audio-dsp COMMAND hudiy-audio-dsp-test)
    add_executable(hudiy-audio-output-test test_audio_output.cpp)
    target_link_libraries(hudiy-audio-output-test PRIVATE Qt6::Core Qt6::Multimedia)
    target_compile_options(hudiy-audio-output-test PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME projection-audio-output COMMAND hudiy-audio-output-test)
    # Explicit silent-sink integration probe; never runs with default CTest.
    add_executable(hudiy-audio-output-stress stress_audio_output.cpp)
    target_link_libraries(hudiy-audio-output-stress PRIVATE Qt6::Core Qt6::Multimedia)
    target_compile_options(hudiy-audio-output-stress PRIVATE -Wall -Wextra -Wpedantic)
endif()
